function doPost(e) { try { var sheet = SpreadsheetApp.getActiveSpreadsheet().getSheetByName("اللاعبين"); var data = JSON.parse(e.postData.contents); var action = data.action; var email = data.email ? data.email.trim().toLowerCase() : ""; var rows = sheet.getDataRange().getValues(); // 1. تسجيل أو جلب بيانات اللاعب if (action === "sync_player") { var found = false; for (var i = 1; i < rows.length; i++) { if (rows[i][2].toString().toLowerCase() === email) { // تحديث المرحلة والرصيد الحالي sheet.getRange(i + 1, 5).setValue(data.maxStage); sheet.getRange(i + 1, 6).setValue(data.coins); found = true; return ContentService.createTextOutput(JSON.stringify({ status: "success", password: rows[i][3], maxStage: rows[i][4], coins: rows[i][5] })).setMimeType(ContentService.MimeType.JSON); } } if (!found) { // إضافة لاعب جديد للشيت var newId = rows.length; sheet.appendRow([newId, data.name, email, data.password || "123456", data.maxStage || 1, data.coins || 150]); return ContentService.createTextOutput(JSON.stringify({ status: "created" })).setMimeType(ContentService.MimeType.JSON); } } // 2. طلب استعادة كلمة المرور وإرسال إيميل تلقائي if (action === "reset_password") { for (var i = 1; i < rows.length; i++) { if (rows[i][2].toString().toLowerCase() === email) { var playerName = rows[i][1]; var newPass = "Fitna-" + Math.floor(1000 + Math.random() * 9000); sheet.getRange(i + 1, 4).setValue(newPass); // تحديث كلمة المرور بالشيت // إرسال الإيميل التلقائي MailApp.sendEmail({ to: email, subject: "استعادة حسابك في لعبة فِطنة", body: "أهلاً بك يا " + playerName + "،\n\nتلقينا طلباً لاستعادة كلمة المرور الخاصة بحسابك في لعبة فِطنة.\n\nكلمة مرورك الجديدة هي: " + newPass + "\n\nيمكنك تسجيل الدخول بها الآن وتغييرها من قائمة الإعدادات داخل اللعبة في أي وقت." }); return ContentService.createTextOutput(JSON.stringify({ status: "success" })).setMimeType(ContentService.MimeType.JSON); } } return ContentService.createTextOutput(JSON.stringify({ status: "not_found" })).setMimeType(ContentService.MimeType.JSON); } // 3. تغيير كلمة المرور من داخل اللعبة if (action === "change_password") { for (var i = 1; i < rows.length; i++) { if (rows[i][2].toString().toLowerCase() === email) { sheet.getRange(i + 1, 4).setValue(data.newPassword); return ContentService.createTextOutput(JSON.stringify({ status: "success" })).setMimeType(ContentService.MimeType.JSON); } } } } catch(error) { return ContentService.createTextOutput(JSON.stringify({ status: "error", message: error.toString() })).setMimeType(ContentService.MimeType.JSON); } }
